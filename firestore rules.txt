rules_version = '2';

service cloud.firestore {

  match /databases/{database}/documents {

    /*
    ==========================================
    SECURITY HELPER FUNCTIONS
    ==========================================
    */

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn() &&
        request.auth.token.admin == true;
    }

    function isSuperAdmin() {
      return isSignedIn() &&
        request.auth.token.superAdmin == true;
    }

    function isOwner(uid) {
      return isSignedIn() &&
        request.auth.uid == uid;
    }

    function unchanged(field) {
      return request.resource.data[field] ==
             resource.data[field];
    }

    function isSellerOwner(sellerId) {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/sellers/$(sellerId)) &&
        get(/databases/$(database)/documents/sellers/$(sellerId)).data.userId == request.auth.uid;
    }

    function isSellerDocumentOwner(documentId) {
      return isSignedIn() &&
        exists(/databases/$(database)/documents/sellers/$(documentId)) &&
        get(/databases/$(database)/documents/sellers/$(documentId)).data.userId == request.auth.uid;
    }

    /*
    ==========================================
    USERS
    ==========================================
    */

    match /users/{uid} {
      allow read: if isOwner(uid) || isAdmin() || isSuperAdmin();
      allow create: if isOwner(uid) && request.resource.data.id == uid;
      allow update: if isOwner(uid)
        && unchanged("id")
        && (!("admin" in request.resource.data) || request.resource.data.admin != true)
        && (!("superAdmin" in request.resource.data) || request.resource.data.superAdmin != true);
      allow delete: if isOwner(uid) || isSuperAdmin();
    }

    /*
    ==========================================
    SELLER PROFILES
    ==========================================
    */
    match /sellers/{uid} {
      allow read: if true;
      allow create: if isSignedIn() && request.resource.data.userId == request.auth.uid;
      allow update: if isSignedIn()
        && resource.data.userId == request.auth.uid
        && request.resource.data.userId == resource.data.userId;
      allow delete: if isSignedIn() && resource.data.userId == request.auth.uid;

      match /customers/{customerId} {
        allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow create: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow update: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      }

      match /suppliers/{supplierId} {
        allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow create: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow update: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      }

      match /repairRequests/{requestId} {
        allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow create: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow update: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      }

      match /stockAdjustments/{adjustmentId} {
        allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow create: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow update: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      }

      match /payoutMethods/{methodId} {
        allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow create: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow update: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      }

      match /purchaseOrders/{orderId} {
        allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow create: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow update: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      }

      match /messages/{messageId} {
        allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
        allow create: if (isSellerOwner(uid) || isSellerDocumentOwner(uid))
          && request.resource.data.senderId == request.auth.uid;
        allow update, delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      }
    }

    /*
    ==========================================
    GENERAL COLLECTIONS
    ==========================================
    */

    match /customers/{customerId} {
      allow read: if isSignedIn();
      allow create: if isSignedIn();
      allow update: if isSignedIn();
      allow delete: if isSuperAdmin();
    }

    match /repairRequests/{requestId} {
      allow read: if isSignedIn() && (
        resource.data.buyerId == request.auth.uid || isSellerOwner(resource.data.sellerId)
      );
      allow create: if isSignedIn() && request.resource.data.buyerId == request.auth.uid;
      allow update: if isSignedIn() && (
        resource.data.buyerId == request.auth.uid || isSellerOwner(resource.data.sellerId)
      );
      allow delete: if isSuperAdmin();
    }

    match /withdrawalRequests/{requestId} {
      allow read: if isSignedIn() && isSellerOwner(resource.data.sellerId);
      allow create: if isSignedIn()
        && exists(/databases/$(database)/documents/sellers/$(request.resource.data.sellerId))
        && get(/databases/$(database)/documents/sellers/$(request.resource.data.sellerId)).data.userId == request.auth.uid;
      allow update: if isSignedIn() && isSellerOwner(resource.data.sellerId);
      allow delete: if isSuperAdmin();
    }

    match /shops/{shopId} {
      allow read: if true;
      allow create: if isSignedIn() && request.resource.data.ownerId == request.auth.uid;
      allow update: if isSignedIn()
        && resource.data.ownerId == request.auth.uid
        && request.resource.data.ownerId == resource.data.ownerId;
      allow delete: if isSuperAdmin() || (isSignedIn() && resource.data.ownerId == request.auth.uid);
    }

    match /products/{productId} {
      allow read: if true;
      allow create: if isSignedIn()
        && request.resource.data.sellerId == request.auth.uid
        && request.resource.data.keys().hasAll(["name", "price", "stock", "sellerId", "shopId", "createdAt"])
        && request.resource.data.price is number
        && request.resource.data.stock is int;
      allow update: if isSignedIn()
        && resource.data.sellerId == request.auth.uid
        && request.resource.data.sellerId == resource.data.sellerId
        && request.resource.data.price is number
        && request.resource.data.stock is int;
      allow delete: if isSuperAdmin() || (isSignedIn() && resource.data.sellerId == request.auth.uid);
    }

    match /sellers/{uid}/products/{productId} {
      allow read: if true;
      allow create: if (isSellerOwner(uid) || isSellerDocumentOwner(uid))
        && request.resource.data.sellerId == uid;
      allow update: if (isSellerOwner(uid) || isSellerDocumentOwner(uid))
        && resource.data.sellerId == uid
        && request.resource.data.sellerId == resource.data.sellerId;
      allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
    }

    match /categories/{categoryId} {
      allow read: if true;
      allow create, update, delete: if isAdmin() || isSuperAdmin();
    }

    match /orders/{orderId} {
      allow read: if isSignedIn() && (
        request.auth.uid == resource.data.buyerId
        || isSellerOwner(resource.data.sellerId)
        || isAdmin()
        || isSuperAdmin()
      );
      allow create: if isSignedIn() && request.resource.data.total is number && request.resource.data.sellerId is string;
      allow update: if isSignedIn() && (
        request.auth.uid == resource.data.buyerId || isSellerOwner(resource.data.sellerId)
      );
      allow delete: if isSuperAdmin();
    }

    match /sellers/{uid}/orders/{orderId} {
      allow read: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      allow create: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      allow update: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
      allow delete: if isSellerOwner(uid) || isSellerDocumentOwner(uid);
    }

    match /payments/{paymentId} {
      allow read: if isSignedIn() && (
        request.auth.uid == resource.data.buyerId || isAdmin() || isSuperAdmin()
      );
      allow create, update, delete: if false;
    }

    match /refunds/{refundId} {
      allow read: if isAdmin() || isSuperAdmin();
      allow create, update, delete: if isSuperAdmin();
    }

    match /chats/{chatId} {
      allow create: if isSignedIn() && request.auth.uid in request.resource.data.participants;
      allow read, update: if isSignedIn() && request.auth.uid in resource.data.participants;
      allow delete: if false;

      match /messages/{messageId} {
        allow read: if isSignedIn() && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;
        allow create: if isSignedIn()
          && request.resource.data.senderId == request.auth.uid
          && request.auth.uid in get(/databases/$(database)/documents/chats/$(chatId)).data.participants;
        allow delete: if isSignedIn() && resource.data.senderId == request.auth.uid;
        allow update: if false;
      }
    }

    match /reviews/{reviewId} {
      allow read: if true;
      allow create: if isSignedIn()
        && request.resource.data.buyerId == request.auth.uid
        && request.resource.data.keys().hasAll(["buyerId", "productId", "rating", "comment", "createdAt"])
        && request.resource.data.rating >= 1
        && request.resource.data.rating <= 5;
      allow update: if isSignedIn()
        && resource.data.buyerId == request.auth.uid
        && request.resource.data.buyerId == resource.data.buyerId;
      allow delete: if isSignedIn() && resource.data.buyerId == request.auth.uid;
    }

    match /wishlist/{uid} {
      allow read, write: if isOwner(uid);
      match /items/{productId} {
        allow read, write: if isOwner(uid);
      }
    }

    match /notifications/{notificationId} {
      allow read: if isSignedIn() && request.auth.uid == resource.data.userId;
      allow create: if isAdmin() || isSuperAdmin();
      allow update: if isSignedIn() && request.auth.uid == resource.data.userId;
      allow delete: if isSuperAdmin();
    }

    match /analytics/{analyticsId} {
      allow read: if isAdmin() || isSuperAdmin();
      allow create, update, delete: if isSuperAdmin();
    }

    match /auditLogs/{logId} {
      allow read: if isSuperAdmin();
      allow create: if isSuperAdmin();
      allow update, delete: if false;
    }

    match /sellerApplications/{applicationId} {
      allow create: if isSignedIn();
      allow read: if isSignedIn() && (resource.data.userId == request.auth.uid || isAdmin() || isSuperAdmin());
      allow update, delete: if isSuperAdmin();
    }

    match /reports/{reportId} {
      allow create: if isSignedIn();
      allow read: if isSignedIn() && (resource.data.reporterId == request.auth.uid || isAdmin() || isSuperAdmin());
      allow update, delete: if isSuperAdmin();
    }

    match /systemSettings/{settingId} {
      allow read: if isAdmin() || isSuperAdmin();
      allow write: if isSuperAdmin();
    }

    match /adminUsers/{adminId} {
      allow read, write: if isSuperAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
